class Manager:
    def __init__(self, name, join_date, salary):
        self.name = name
        self.join_date = join_date
        self._salary = salary
    def getmanagersalary(self, obj):
        print("Since salary is a protected attribute, it did not change")
        print(obj._salary)

    def print_all(self):
        print("::Manager Information::")
        print(self.name)
        print(self.join_date)
        print(self._salary)
        print("---------------------")

class CEO(Manager):
    def __init__(self, name, join_date, salary):
        self.name = name
        self.join_date = join_date
        self.__salary = salary
    def setmanagersalary(self, obj, newsalary):
        obj._salary = newsalary

    def setCEOsalary(self, obj, newsalary):
        obj.__salary = newsalary
    def getCEOsalary(self, obj):
        print("New salary")
        print(obj.__salary)
    
    def print_all(self):
        print("::CEO Information::")
        print(self.name)
        print(self.join_date)
        print(self.__salary)
        print("---------------------")
