class Employee:
   def __init__(self, name, salary):
      self.name = name
      self.__salary = salary
   
   @property
   def salary(self):
     return self.__salary
   
   @salary.setter
   def salary(self, value):
      if value >= 0:
         self.__salary = value
      else:
         raise ValueError("Salary cannot be negative")

emp1 = Employee("Nahid", 50000)
print(emp1.salary)
emp1.salary=40000
print(emp1.salary)
